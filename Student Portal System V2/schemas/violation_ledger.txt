create table public.violation_ledger (
  violation_id bigint generated by default as identity not null,
  source_attendance_id integer null,
  student_pk_id integer null,
  student_id character varying(15) not null,
  student_name character varying(100) not null,
  program character varying(150) null,
  year_level smallint null,
  event_id integer null,
  event_name character varying(150) not null,
  event_date date not null,
  time_in timestamp without time zone null,
  time_out timestamp without time zone null,
  violation_status character varying(30) not null,
  assessed_fine numeric(10, 2) not null default 0,
  fine_paid numeric(10, 2) not null default 0,
  fine_balance numeric GENERATED ALWAYS as (
    GREATEST((assessed_fine - fine_paid), (0)::numeric)
  ) STORED (10, 2) null,
  assessed_community_service_hours numeric(6, 1) not null default 0,
  community_service_hours_rendered numeric(6, 1) not null default 0,
  community_service_hours_balance numeric GENERATED ALWAYS as (
    GREATEST(
      (
        assessed_community_service_hours - community_service_hours_rendered
      ),
      (0)::numeric
    )
  ) STORED (6, 1) null,
  is_excused boolean not null default false,
  excused_reason character varying(255) null,
  migrated_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  updated_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  is_active_violation boolean not null default true,
  excuse_request_id integer null,
  excuse_request_status character varying(10) null,
  excuse_request_reason character varying(255) null,
  excuse_letter_path character varying(255) null,
  supporting_document_path character varying(255) null,
  excuse_decision_notes character varying(255) null,
  excuse_requested_at timestamp without time zone null,
  excuse_decided_at timestamp without time zone null,
  excuse_requested_by integer null,
  excuse_reviewed_by integer null,
  public_student_id uuid null,
  public_event_id uuid null,
  public_event_attendance_id uuid null,
  waived_fine numeric(10, 2) not null default 0,
  waived_community_service_hours numeric(6, 2) not null default 0,
  effective_fine_balance numeric GENERATED ALWAYS as (
    GREATEST(
      (
        (
          COALESCE(assessed_fine, (0)::numeric) - COALESCE(fine_paid, (0)::numeric)
        ) - COALESCE(waived_fine, (0)::numeric)
      ),
      (0)::numeric
    )
  ) STORED (10, 2) null,
  effective_community_service_hours_balance numeric GENERATED ALWAYS as (
    GREATEST(
      (
        (
          COALESCE(assessed_community_service_hours, (0)::numeric) - COALESCE(community_service_hours_rendered, (0)::numeric)
        ) - COALESCE(waived_community_service_hours, (0)::numeric)
      ),
      (0)::numeric
    )
  ) STORED (6, 2) null,
  constraint violation_ledger_pkey primary key (violation_id),
  constraint violation_ledger_excuse_request_id_key unique (excuse_request_id),
  constraint violation_ledger_source_attendance_id_key unique (source_attendance_id),
  constraint public_violation_ledger_nonnegative_amounts check (
    (
      (assessed_fine >= (0)::numeric)
      and (fine_paid >= (0)::numeric)
      and (assessed_community_service_hours >= (0)::numeric)
      and (community_service_hours_rendered >= (0)::numeric)
    )
  ),
  constraint violation_ledger_nonnegative_waivers check (
    (
      (waived_fine >= (0)::numeric)
      and (waived_community_service_hours >= (0)::numeric)
    )
  )
) TABLESPACE pg_default;

create index IF not exists public_violation_ledger_public_student_idx on public.violation_ledger using btree (public_student_id) TABLESPACE pg_default;

create unique INDEX IF not exists uq_violation_ledger_public_event_attendance on public.violation_ledger using btree (public_event_attendance_id) TABLESPACE pg_default
where
  (public_event_attendance_id is not null);

create index IF not exists idx_violation_ledger_public_event on public.violation_ledger using btree (public_event_id) TABLESPACE pg_default
where
  (public_event_id is not null);

create index IF not exists public_violation_ledger_student_event_idx on public.violation_ledger using btree (student_pk_id, event_id) TABLESPACE pg_default;

create index IF not exists public_violation_ledger_outstanding_idx on public.violation_ledger using btree (event_date desc) TABLESPACE pg_default
where
  (
    (fine_balance > (0)::numeric)
    or (community_service_hours_balance > (0)::numeric)
  );

create trigger audit_ebams_violation_ledger
after INSERT
or DELETE
or
update on violation_ledger for EACH row
execute FUNCTION audit_ebams_system_activity ('violation_ledger');

create trigger link_public_violation_ledger_student BEFORE INSERT
or
update OF student_pk_id on violation_ledger for EACH row
execute FUNCTION link_violation_ledger_student ();
