create table public.events (
  id uuid not null default extensions.gen_random_uuid (),
  title text not null,
  event_date date not null,
  required_time_in time without time zone null,
  required_time_out time without time zone null,
  late_threshold_minutes integer not null default 15,
  venue text null,
  is_afternoon boolean not null default false,
  allow_time_in boolean not null default true,
  allow_time_out boolean not null default false,
  attendance_locked boolean not null default false,
  is_active boolean not null default true,
  created_by uuid null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  attendance_finalized_at timestamp with time zone null,
  archived_at timestamp with time zone null,
  archived_by uuid null,
  constraint events_pkey primary key (id)
) TABLESPACE pg_default;

create index IF not exists idx_events_completed_schedule on public.events using btree (event_date desc, required_time_in desc) TABLESPACE pg_default
where
  (
    (required_time_in is not null)
    and (required_time_out is not null)
  );

create index IF not exists idx_events_date on public.events using btree (event_date) TABLESPACE pg_default;

create index IF not exists idx_events_active on public.events using btree (is_active) TABLESPACE pg_default;

create index IF not exists idx_events_attendance_finalized on public.events using btree (attendance_finalized_at) TABLESPACE pg_default
where
  (attendance_finalized_at is not null);

create index IF not exists idx_events_active_unarchived on public.events using btree (event_date desc, required_time_in desc) TABLESPACE pg_default
where
  (archived_at is null);

create index IF not exists idx_events_active_schedule on public.events using btree (event_date, required_time_in) TABLESPACE pg_default
where
  (
    (is_active = true)
    and (required_time_in is not null)
    and (required_time_out is not null)
  );

create trigger audit_ebams_events
after INSERT
or DELETE
or
update on events for EACH row
execute FUNCTION audit_ebams_system_activity ('event');
