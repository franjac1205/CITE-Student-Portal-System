-- Student Portal System V2
-- PostgreSQL / Supabase compatible DDL

-- Supabase installs extensions into a dedicated "extensions" schema by
-- default (not "public"), so pgcrypto's functions must be called as
-- extensions.crypt(...) / extensions.gen_salt(...) unless that schema is
-- on the search_path.
create extension if not exists "pgcrypto" with schema extensions;

-- =========================================================
-- USERS
-- =========================================================
create table if not exists public.users (
  id text primary key default gen_random_uuid()::text,
  student_id text not null unique,
  first_name text not null,
  last_name text not null,
  password text not null,          -- stored as a bcrypt hash via pgcrypto, never plaintext
  role text not null check (role in ('student','officer','president','finance')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.users enable row level security;

-- Supabase exposes this table over the API by default, so RLS must be on.
-- The app needs to read the users table for the dashboard and profile views,
-- and it needs to update profile values from the client. Keep direct inserts
-- blocked and use the secure login function for authentication.
drop policy if exists "no client access to users" on public.users;
drop policy if exists "allow client read access to users" on public.users;
drop policy if exists "allow client update access to users" on public.users;
drop policy if exists "no client access to users inserts" on public.users;
drop policy if exists "no client access to users deletes" on public.users;
create policy "allow client read access to users"
  on public.users
  for select
  using (true);
create policy "allow client update access to users"
  on public.users
  for update
  using (true)
  with check (true);
create policy "no client access to users inserts"
  on public.users
  for insert
  with check (false);
create policy "no client access to users deletes"
  on public.users
  for delete
  using (false);

grant select on public.users to anon, authenticated;
grant update on public.users to authenticated;

-- Runs with the privileges of its owner (bypassing RLS above), so it alone
-- can compare the password hash without ever exposing the password column
-- to the client directly. Return columns changed across edits, so drop
-- the function first -- CREATE OR REPLACE cannot alter OUT parameters.
drop function if exists public.check_login(text, text);
create or replace function public.check_login(p_student_id text, p_password text)
returns table (id text, student_id text, first_name text, last_name text, role text)
language sql
security definer
set search_path = public, extensions
as $$
  select id, student_id, first_name, last_name, role
  from public.users
  where student_id = p_student_id
    and password = extensions.crypt(p_password, password)
$$;

drop function if exists public.update_user_profile(text, text, text, text);
create or replace function public.update_user_profile(p_user_id text, p_first_name text, p_last_name text, p_password text)
returns table (id text, student_id text, first_name text, last_name text, password text, role text)
language plpgsql
security definer
set search_path = public, extensions
as $$
begin
  if p_password is null or p_password = '' then
    update public.users
    set first_name = p_first_name,
        last_name = p_last_name,
        updated_at = now()
    where id = p_user_id;
  else
    update public.users
    set first_name = p_first_name,
        last_name = p_last_name,
        password = extensions.crypt(p_password, extensions.gen_salt('bf')),
        updated_at = now()
    where id = p_user_id;
  end if;

  return query
  select id, student_id, first_name, last_name, password, role
  from public.users
  where id = p_user_id;
end;
$$;

-- Let any client call the functions itself (the function body is what
-- restricts what data comes back), but keep direct table access closed.
grant execute on function public.check_login(text, text) to anon, authenticated;
grant execute on function public.update_user_profile(text, text, text, text) to anon, authenticated;

-- FINE SETTINGS
-- =========================================================
create table if not exists public.fine_settings (
  id bigint generated always as identity primary key,
  singleton boolean not null default true unique, -- enforces exactly one row
  late integer not null default 100,
  absent integer not null default 200,
  wrong_attire integer not null default 50,
  updated_at timestamptz not null default now(),
  constraint fine_settings_singleton check (singleton)
);

alter table public.fine_settings enable row level security;

drop policy if exists "anyone can read fine settings" on public.fine_settings;

  with check (false);
